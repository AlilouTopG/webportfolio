# Ziegler-Nichols Ultimate Gain PID Auto-Tuning
import nexus as nx

plc = nx.ModbusTCP("192.168.10.50:502")
broker = nx.MQTT("tls://nexus.mqtt:8883")

with nx.DigitalTwin("reactor-04") as twin:
    twin.bind(plc, broker, rbac="supabase")
    pid = twin.auto_tune(method="ziegler_nichols")
    # → Ku=4.82  Tu=1.34s  Kp=2.89 Ki=4.31 Kd=0.48
    twin.deploy(pid, strategy="canary")
    print(f"✓ Deployed PID {pid.id} | overshoot < 4.2%")
